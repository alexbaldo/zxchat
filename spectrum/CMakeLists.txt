cmake_minimum_required(VERSION 3.16)

# The Spectranext SDK is imported BEFORE project()
include($ENV{SPECTRANEXT_SDK_PATH}/cmake/spectranext_sdk_import.cmake)
spectranext_sdk_init()

project(zxchat C)

# The version comes from the VERSION file at the root, so it is not written in
# two places that can drift apart. It goes into a generated header rather than
# the command line: quotes are lost there and the literal arrives broken.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" ZXCHAT_VERSION)
string(STRIP "${ZXCHAT_VERSION}" ZXCHAT_VERSION)
file(WRITE "${CMAKE_BINARY_DIR}/version.h"
     "/* Generated by CMake from the VERSION file. */\n"
     "#define ZXCHAT_VERSION \"${ZXCHAT_VERSION}\"\n")
include_directories("${CMAKE_BINARY_DIR}")

# Interface language. Only one is compiled in: two sets of strings and two
# prompts do not fit in what is left of 48K.
set(ZXCHAT_LANG "es" CACHE STRING "es | en")
if(ZXCHAT_LANG STREQUAL "en")
    add_compile_definitions(CFG_LANG_EN=1)
endif()

add_executable(zxchat
    src/zxchat.c     # the BASIC extension: the three modes
    src/direct.c     # talks to the API, reads the SSE stream
    src/tokenize.c   # BASIC text -> tokens
    src/screen.c     # 32-column renderer, drawn straight to video memory
    src/input.c      # keyboard, through the ROM interrupt
    src/wrap.c       # word wrapping and UTF-8 down to ASCII
)

target_include_directories(zxchat PUBLIC src)

# sccz80's -debug wraps every function in a stack frame (call l_debug_push_frame
# and pop_frame) and emits C_LINE and CDBINFO symbols. That is around 1.6K on a
# machine that has none to spare, so it is off by default. Turn it back on with
# -DZXCHAT_DEBUG=ON when something needs stepping through.
option(ZXCHAT_DEBUG "Build with sccz80 debug frames and line info" OFF)
if(ZXCHAT_DEBUG)
    target_compile_options(zxchat PUBLIC -debug)
    target_link_options(zxchat PUBLIC -debug)
endif()

# The _np libraries page nothing by themselves, which is what a BASIC
# extension needs: the normal ones leave the cartridge paged in a way that
# does not survive the jump back to BASIC. That also rules out libhttp, which
# has no _np variant, so the API is reached with raw sockets.
target_link_libraries(zxchat PUBLIC
    -lndos
    -llibspectranet_np
    -llibsocket_np
)
target_link_options(zxchat PUBLIC -create-app)
